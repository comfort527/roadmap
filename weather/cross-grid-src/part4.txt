  async function json(url,signal,timeoutMs=25000){
    const ctl=new AbortController(),timer=setTimeout(()=>ctl.abort(),timeoutMs),abort=()=>ctl.abort();
    if(signal){if(signal.aborted)ctl.abort();else signal.addEventListener('abort',abort,{once:true})}
    try{
      const res=await fetch(url,{signal:ctl.signal});
      if(!res.ok){let reason='';try{const body=await res.json();reason=body?.reason||body?.error||''}catch{}throw new Error(reason?`服務暫時無法回應（${res.status}：${reason}）`:`服務暫時無法回應（${res.status}）`)}
      return res.json();
    }finally{clearTimeout(timer);if(signal)signal.removeEventListener('abort',abort)}
  }
